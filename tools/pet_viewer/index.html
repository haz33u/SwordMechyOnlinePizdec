💡 Как сделать 3D-рендеры петов в Roblox Studio сочными и реалистичными

В Roblox 3D-модели петов внутри ViewportFrame часто выглядят блекло, если не настроить правильную камеру и студийное освещение. Ниже приведена готовая функция для Luau, которая делает 3D-петы в инвентаре яркими, объемными и вращающимися: 

-- High-Quality Studio ViewportFrame Renderer for Roblox
local function SetupStudioViewport(viewportFrame: ViewportFrame, petModel: Model)
    viewportFrame.Ambient = Color3.fromRGB(180, 190, 210) -- Мягкий заполняющий свет
    viewportFrame.LightColor = Color3.fromRGB(255, 245, 230) -- Тёплый студийный свет
    viewportFrame.LightDirection = Vector3.new(-1, -1.5, -1).Unit -- Верхне-боковой угол

    local camera = Instance.new("Camera")
    camera.FieldOfView = 25 -- Узкий FOV для глубокого студийного объёма
    viewportFrame.CurrentCamera = camera
    camera.Parent = viewportFrame

    local cf, size = petModel:GetBoundingBox()
    local maxExtent = math.max(size.X, size.Y, size.Z)
    local dist = maxExtent * 2.2

    camera.CFrame = CFrame.lookAt(
        cf.Position + Vector3.new(dist * 0.4, dist * 0.3, dist),
        cf.Position
    )
end
